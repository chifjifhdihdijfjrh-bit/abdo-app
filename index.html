import os import uuid import shutil import logging import tempfile from pathlib import Path import whisper from fastapi import FastAPI, UploadFile, File, HTTPException from fastapi.responses import FileResponse from starlette.background import BackgroundTask from moviepy import VideoFileClip, TextClip, CompositeVideoClip # ===================================================== # Abdo AI Video Engine Pro - Version 4.1 # Fixed Auto-Zoom + Arabic Captions + MP4 Export # ===================================================== logging.basicConfig(level=logging.INFO) logger = logging.getLogger("abdo-ai-video") APP_NAME = "Abdo AI Video Engine Pro" APP_VERSION = "4.1" MAX_UPLOAD_MB = int(os.getenv("MAX_UPLOAD_MB", "300")) MAX_UPLOAD_BYTES = MAX_UPLOAD_MB * 1024 * 1024 WHISPER_MODEL = os.getenv("WHISPER_MODEL", "base") OUTPUT_FPS = int(os.getenv("OUTPUT_FPS", "24")) ZOOM_FACTOR = float(os.getenv("ZOOM_FACTOR", "1.15")) ARABIC_FONT_PATH = os.getenv("ARABIC_FONT_PATH", "") TEMP_ROOT = os.getenv("VIDEO_TEMP_DIR", tempfile.gettempdir()) ALLOWED_EXTENSIONS = { ".mp4", ".mov", ".m4v", ".webm", ".avi", ".mkv" } app = FastAPI( title=APP_NAME, version=APP_VERSION, description="AI transcription, Arabic captions, auto-zoom and MP4 export" ) model = None @app.on_event("startup") def load_whisper_model(): global model logger.info("Loading Whisper model: %s", WHISPER_MODEL) try: model = whisper.load_model(WHISPER_MODEL) logger.info("Whisper model loaded successfully") except Exception: logger.exception("Whisper model could not be loaded") raise @app.get("/") def home(): return { "status": "online", "service": APP_NAME, "version": APP_VERSION, "creator": "Abdo", "features": [ "speech_transcription", "timed_captions", "arabic_caption_support", "center_crop_auto_zoom", "mp4_export" ] } @app.get("/health") def health(): return { "status": "ready" if model is not None else "loading", "model": WHISPER_MODEL, "version": APP_VERSION } def cleanup_directory(path): if path: shutil.rmtree(path, ignore_errors=True) def close_clip(clip): if clip is not None: try: clip.close() except Exception: logger.warning("Could not close a video resource", exc_info=True) def prepare_arabic_text(text): """ تجهيز النص العربي للعرض. يحتاج arabic-reshaper و python-bidi. """ try: import arabic_reshaper from bidi.algorithm import get_display return get_display(arabic_reshaper.reshape(text)) except ImportError: logger.warning( "Arabic shaping packages are missing; using original text" ) return text except Exception: logger.exception("Arabic text shaping failed") return text def get_font_path(): candidates = [ ARABIC_FONT_PATH, "/usr/share/fonts/truetype/cairo/Cairo-Regular.ttf", "/usr/share/fonts/truetype/noto/NotoSansArabic-Regular.ttf", "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf", "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", "C:/Windows/Fonts/arial.ttf", "C:/Windows/Fonts/tahoma.ttf", ] for candidate in candidates: if candidate and Path(candidate).is_file(): return str(Path(candidate).resolve()) raise RuntimeError( "Arabic font not found. Set ARABIC_FONT_PATH " "to the full path of an installed font file." ) def make_zoom_clip(video, zoom_factor): """ تكبير الفيديو ثم قص الزوائد من المنتصف للحفاظ على أبعاد الفيديو الأصلية. """ if zoom_factor < 1.0 or zoom_factor > 2.0: raise ValueError("ZOOM_FACTOR must be between 1.0 and 2.0") if zoom_factor == 1.0: return video zoomed = video.resized(zoom_factor) # قص الجزء الأوسط بعد التكبير وإرجاع الأبعاد الأصلية. zoomed = zoomed.cropped( x_center=zoomed.w / 2, y_center=zoomed.h / 2, width=video.w, height=video.h ) return zoomed.with_duration(video.duration) def make_caption(text, start, duration, video, font_path): caption_width = max(1, int(video.w * 0.85)) font_size = max(22, min(54, int(video.w * 0.04))) caption = TextClip( text=prepare_arabic_text(text), font=font_path, font_size=font_size, color="white", stroke_color="black", stroke_width=2, method="caption", size=(caption_width, None), text_align="center", horizontal_align="center", margin=(12, 8) ) return ( caption .with_start(start) .with_duration(duration) .with_position(("center", int(video.h * 0.76))) ) @app.post("/process-video/") async def process_video(file: UploadFile = File(...)): if model is None: raise HTTPException( status_code=503, detail="AI model is not ready. Try again shortly." ) original_name = file.filename or "video.mp4" extension = Path(original_name).suffix.lower() if extension not in ALLOWED_EXTENSIONS: raise HTTPException( status_code=415, detail="Unsupported video format." ) try: os.makedirs(TEMP_ROOT, exist_ok=True) work_dir = tempfile.mkdtemp( prefix="abdo_video_", dir=TEMP_ROOT ) except OSError: logger.exception("Could not create temporary directory") raise HTTPException( status_code=500, detail="Could not prepare video processing storage." ) input_path = os.path.join( work_dir, f"{uuid.uuid4().hex}{extension}" ) output_path = os.path.join( work_dir, f"{uuid.uuid4().hex}_edited.mp4" ) video = None zoomed_video = None final_video = None caption_clips = [] try: # --------------------------------------------- # 1. حفظ الفيديو والتحقق من حجم الملف # --------------------------------------------- total_bytes = 0 with open(input_path, "wb") as output_file: while True: chunk = await file.read(1024 * 1024) if not chunk: break total_bytes += len(chunk) if total_bytes > MAX_UPLOAD_BYTES: raise HTTPException( status_code=413, detail=f"Video exceeds the {MAX_UPLOAD_MB} MB limit." ) output_file.write(chunk) if total_bytes == 0: raise HTTPException( status_code=400, detail="The uploaded video is empty." ) # --------------------------------------------- # 2. تفريغ الكلام باستخدام Whisper # --------------------------------------------- logger.info("Transcribing uploaded video") result = model.transcribe( input_path, task="transcribe", word_timestamps=True, verbose=False ) segments = result.get("segments", []) # --------------------------------------------- # 3. فتح الفيديو والخط العربي # --------------------------------------------- video = VideoFileClip(input_path) if video.duration <= 0: raise HTTPException( status_code=400, detail="The uploaded video has no valid duration." ) font_path = get_font_path() # --------------------------------------------- # 4. تطبيق الزووم فعليًا على الفيديو الأساسي # --------------------------------------------- zoomed_video = make_zoom_clip(video, ZOOM_FACTOR) # --------------------------------------------- # 5. إنشاء الترجمة المتزامنة مع الكلام # --------------------------------------------- for segment in segments: text = str(segment.get("text", "")).strip() start = max( 0.0, float(segment.get("start", 0.0)) ) end = min( float(video.duration), float(segment.get("end", start)) ) duration = end - start if not text or duration <= 0: continue caption = make_caption( text=text, start=start, duration=duration, video=video, font_path=font_path ) caption_clips.append(caption) # --------------------------------------------- # 6. دمج الفيديو المكبّر والترجمة # --------------------------------------------- final_video = CompositeVideoClip( [zoomed_video] + caption_clips, size=video.size ).with_duration(video.duration) # الاحتفاظ بصوت الفيديو الأصلي عند وجوده if zoomed_video.audio is not None: final_video = final_video.with_audio(zoomed_video.audio) # --------------------------------------------- # 7. تصدير الفيديو النهائي بصيغة MP4 # --------------------------------------------- logger.info("Rendering final MP4 video") final_video.write_videofile( output_path, fps=OUTPUT_FPS, codec="libx264", audio_codec="aac", preset="medium", threads=2, logger=None, ffmpeg_params=["-movflags", "+faststart"] ) if not os.path.isfile(output_path): raise RuntimeError("MP4 output was not created.") if os.path.getsize(output_path) == 0: raise RuntimeError("The exported MP4 file is empty.") logger.info("Video processing completed successfully") # إغلاق موارد الفيديو قبل إرسال الملف close_clip(final_video) final_video = None # إغلاق مقاطع الترجمة for caption in caption_clips: close_clip(caption) caption_clips.clear() close_clip(zoomed_video) zoomed_video = None close_clip(video) video = None return FileResponse( path=output_path, media_type="video/mp4", filename="abdo_zoom_pro_edited.mp4", background=BackgroundTask( cleanup_directory, work_dir ) ) except HTTPException: cleanup_directory(work_dir) raise except Exception: logger.exception("Video processing failed") cleanup_directory(work_dir) raise HTTPException( status_code=500, detail=( "Video processing failed. Check the server logs " "for the technical error." ) ) finally: # تنظيف الموارد حتى عند فشل المعالجة close_clip(final_video) for caption in caption_clips: close_clip(caption) close_clip(zoomed_video) close_clip(video) try: await file.close() except Exception: pass if __name__ == "__main__": import uvicorn uvicorn.run( app, host="0.0.0.0", port=int(os.getenv("PORT", "8000")) )
